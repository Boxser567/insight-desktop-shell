import { describe, expect, it, vi } from 'vitest'
import { RemoteWorkflowBackend } from '../packages/insight-workflow-op/src/remote-workflow'
import { CanvasRequestError, type CanvasApi } from '../packages/insight-workflow-op/src/canvas-api'
import { workflowSchemas } from '../packages/insight-workflow-protocol/src/workflow-schemas'

// Review-only probes. All API calls are injected fakes; no real account or network.
function setup() {
  const makeNode = (id: string) => ({ id, type: 'video', position: { x: 0, y: 0 }, data: {
    title: id, value: '', params: { model: 'v', modelType: 'text_to_video', prompt: 'old' }
  } })
  const flow: any = { id: 42, name: 'Review', version: 'comfyui', versionNumber: 3, canvasType: 0,
    data: { nodes: [makeNode('a'), makeNode('b')], edges: [] } }
  const config: any = { video: { text_to_video: { models: { v: {
    model_name: 'Video', parameters: {
      duration: { type: 'number', label: 'seconds', min: 1, max: 10, step: 1, default: 5 }
    }
  } } } } }
  const updateCanvas = vi.fn(async (update: any) => {
    flow.data.nodes.push(...(update.addNodes ?? []).map((s: string) => JSON.parse(s)))
    for (const value of update.updateNodes ?? []) {
      const n = JSON.parse(value)
      flow.data.nodes[flow.data.nodes.findIndex((x: any) => x.id === n.id)] = n
    }
    flow.data.nodes = flow.data.nodes.filter((n: any) => !update.deleteNodeIds?.includes(n.id))
    if (update.edges) flow.data.edges = update.edges
    flow.versionNumber++
    return { code: 'SUCCESS', data: { id: 42 } }
  })
  const api = {
    getFlowData: vi.fn(async () => ({ code: 'SUCCESS', data: structuredClone(flow) })),
    getModelParams: vi.fn(async () => config), updateCanvas,
    saveFlow: vi.fn(async () => ({ code: 'SUCCESS', data: { id: 42 } })),
    getFlowsList: vi.fn(), dispose: vi.fn()
  } as unknown as CanvasApi
  const backend = new RemoteWorkflowBackend(api)
  const patch = (operations: unknown[]) => backend.execute('apply_workflow_patch',
    workflowSchemas.apply_workflow_patch.input.parse({ workflowId: '42', operations }))
  return { flow, config, api, backend, patch, updateCanvas }
}

describe('77fb977 contract and real remote adapter review observations', () => {
  it('R01 no longer accepts temporary refs, dryRun or patch idempotency; still has no expectedVersion/text', () => {
    const schema = workflowSchemas.apply_workflow_patch.input
    for (const extra of [{ dryRun: true }, { idempotencyKey: 'x' }, { expectedVersion: '3' }]) {
      expect(schema.safeParse({ workflowId: '42', operations: [], ...extra }).success).toBe(false)
    }
    expect(schema.safeParse({ workflowId: '42', operations: [
      { op: 'add_node', clientId: 'new', node: { type: 'text', model: 't' } }
    ] }).success).toBe(false)
    expect(schema.safeParse({ workflowId: '42', operations: [
      { op: 'add_node', node: { type: 'text', model: 't', text: 'written copy' } }
    ] }).success).toBe(false)
  })

  it('R02 optional handles pass schema but remote rejects; no ports supplied by discovery', async () => {
    const { patch, backend } = setup()
    expect(await patch([{ op: 'add_edge', source: 'a', target: 'b' }])).toMatchObject({ applied: false })
    const definition = await backend.execute('get_node_definition', { nodeType: 'video' })
    expect(definition.nodeDefinitions[0]).not.toHaveProperty('ports')
  })

  it('R03 explicit nonexistent handles, duplicate edges and cycle reach updateCanvas', async () => {
    const { patch, updateCanvas } = setup()
    const edge = { op: 'add_edge', source: 'a', target: 'b', sourceHandle: 'invented', targetHandle: 'invented' }
    expect(await patch([edge, edge, { ...edge, source: 'b', target: 'a' }])).toMatchObject({ applied: true })
    expect(updateCanvas.mock.calls[0][0].edges).toHaveLength(3)
  })

  it('R04 native add payload contains overlapping positions and running state without execution', async () => {
    const { patch, updateCanvas } = setup()
    const add = { op: 'add_node', node: { type: 'video', model: 'v', modelType: 'text_to_video' } }
    expect(await patch([add, add])).toMatchObject({ applied: true })
    const saved = updateCanvas.mock.calls[0][0].addNodes.map((s: string) => JSON.parse(s))
    expect(saved.map((n: any) => n.position)).toEqual([{ x: 0, y: 0 }, { x: 0, y: 0 }])
    expect(saved[0].data.taskInfo).toEqual({ status: 'running' })
  })

  it('R05 model duration max 10 is discoverable but duration 999 reaches save', async () => {
    const { patch, updateCanvas } = setup()
    expect(await patch([{ op: 'update_node', id: 'a', changes: { type: 'video', duration: 999 } }]))
      .toMatchObject({ applied: true })
    expect(JSON.parse(updateCanvas.mock.calls[0][0].updateNodes[0]).data.params.duration).toBe(999)
  })

  it('R06 reads latest backend version without checking the snapshot that Agent used', async () => {
    const { backend, flow, patch, updateCanvas } = setup()
    const old = await backend.execute('get_node', { workflowId: '42', id: 'a' })
    flow.versionNumber = 4
    flow.data.nodes[0].data.params.prompt = 'user newer edit'
    await patch([{ op: 'update_node', id: 'a', changes: { type: 'video', prompt: old.node.data.params.prompt + ' agent edit' } }])
    expect(updateCanvas.mock.calls[0][0].versionNumber).toBe(4)
    expect(JSON.parse(updateCanvas.mock.calls[0][0].updateNodes[0]).data.params.prompt).toBe('old agent edit')
  })

  it('R07 persisted write with lost response throws generic error; retry creates another node', async () => {
    const { patch, updateCanvas, flow } = setup()
    const persist = updateCanvas.getMockImplementation()!
    updateCanvas.mockImplementationOnce(async update => { await persist(update); throw new CanvasRequestError('REQUEST_FAILED') })
    const add = { op: 'add_node', node: { type: 'video', model: 'v', modelType: 'text_to_video' } }
    await expect(patch([add])).rejects.toMatchObject({ code: 'REQUEST_FAILED' })
    expect(flow.data.nodes).toHaveLength(3)
    expect(await patch([add])).toMatchObject({ applied: true })
    expect(flow.data.nodes).toHaveLength(4)
  })

  it('R08 incompatible unrelated basic node prevents targeted get and edit', async () => {
    const { flow, backend, patch, updateCanvas } = setup()
    flow.data.nodes.push({ id: 'legacy', type: 'image', position: { x: 1, y: 1 },
      data: { title: 'legacy', params: { model: 'retired' } } })
    await expect(backend.execute('get_node', { workflowId: '42', id: 'a' })).rejects.toThrow('legacy')
    await expect(patch([{ op: 'update_node', id: 'a', changes: { type: 'video', title: 'new' } }])).rejects.toThrow('legacy')
    expect(updateCanvas).not.toHaveBeenCalled()
  })

  it('R09 numeric choices are silently removed from model metadata', async () => {
    const { config, backend } = setup()
    config.video.text_to_video.models.v.parameters.gen_count = {
      type: 'choices', label: 'count', choices: [{ label: 'one', value: 1 }], default: 1
    }
    const result = await backend.execute('get_model_setting', { query: { filters: { all: [
      { field: 'nodeType', operator: 'eq', value: 'video' },
      { field: 'modelType', operator: 'eq', value: 'text_to_video' }
    ] } } })
    expect(result.model.parameters).not.toHaveProperty('gen_count')
  })

  it('R10 graph patch preserves unprojected fields and separates rename', async () => {
    const { flow, patch, updateCanvas } = setup()
    flow.data.nodes[0].parentId = 'group'
    flow.data.nodes[0].data.extraNativeField = 'keep'
    flow.data.nodes[0].data.params.nativeExtra = { keep: true }
    await patch([{ op: 'update_node', id: 'a', changes: { type: 'video', title: 'new' } }])
    expect(JSON.parse(updateCanvas.mock.calls[0][0].updateNodes[0])).toMatchObject({
      parentId: 'group', data: { extraNativeField: 'keep', params: { nativeExtra: { keep: true } } }
    })
    expect(await patch([{ op: 'rename_workflow', name: 'new' }, { op: 'delete_node', id: 'b' }]))
      .toMatchObject({ applied: false, errors: [{ code: 'UNSUPPORTED_PATCH' }] })
    expect(updateCanvas).toHaveBeenCalledTimes(1)
  })
})
