import { describe, expect, it } from 'vitest'
import { workflowSchemas, workflowTypes } from '../packages/insight-workflow-protocol/src/workflow-schemas'
import { parseCanvasOperation, parseCanvasOperationResult, parseNodeChanges, parseNodeConfiguration, parseNodeDraft } from '/Users/boxser.shi/Documents/harness/desktop-canvas-plugin/src/shared/canvas-agent-protocol'

// Protocol-only probes: no Mock backend, network, Store or media execution.
describe('protocol expression comparison', () => {
  it('W01 workflow expresses typed batch creation and created-node edge references', () => {
    expect(workflowSchemas.apply_workflow_patch.input.safeParse({ workflowId: '103161', operations: [
      { op: 'add_node', clientId: 'copy', node: { type: 'text', model: 'discovered-model' } },
      { op: 'add_node', clientId: 'hero', node: { type: 'image', model: 'discovered-image', modelType: 'reference_image_to_image' } },
      { op: 'add_edge', clientId: 'e', source: { kind: 'created', clientId: 'copy' }, target: { kind: 'created', clientId: 'hero' } }
    ], dryRun: true }).success).toBe(true)
  })
  it('W02 alpha can create literal text without a model; workflow cannot', () => {
    expect(() => parseNodeDraft({ type: 'text', changes: { text: '耳机卖点' } })).not.toThrow()
    expect(workflowTypes.nodeCreate.safeParse({ type: 'text' }).success).toBe(false)
    expect(workflowTypes.nodeCreate.safeParse({ type: 'text', model: 'm', text: '耳机卖点' }).success).toBe(false)
  })
  it('W03 workflow supports nested parameter patches and unset; alpha is scalar only', () => {
    expect(workflowTypes.nodeChanges.safeParse({ type: 'image', additionalParameters: { set: { tuning: { strength: 0.5 } }, unset: ['old'] } }).success).toBe(true)
    expect(() => parseNodeChanges({ params: { tuning: { strength: 0.5 } } })).toThrow()
  })
  it('W04 workflow create lacks additionalParameters supported by its read/update DTO', () => {
    expect(workflowTypes.nodeCreate.safeParse({ type: 'image', model: 'm', modelType: 'text_to_image', additionalParameters: { seed: 7 } }).success).toBe(false)
    expect(workflowTypes.nodeChanges.safeParse({ type: 'image', additionalParameters: { set: { seed: 7 } } }).success).toBe(true)
  })
  it('W05 workflow choices only accept strings', () => {
    expect(workflowTypes.parameter.safeParse({ type: 'choices', label: '数量', choices: [{ value: 1, label: '一张' }] }).success).toBe(false)
    expect(() => parseNodeConfiguration({ params: { gen_count: 1, enabled: true } })).not.toThrow()
  })
  it('W06 alpha accepts discovered mode identifiers while workflow uses closed enums', () => {
    expect(() => parseNodeConfiguration({ modelType: 'future_discovered_mode' })).not.toThrow()
    expect(workflowTypes.nodeCreate.safeParse({ type: 'image', model: 'm', modelType: 'future_discovered_mode' }).success).toBe(false)
  })
  it('W07 queued execution reply cannot round trip through workflow run query schema', () => {
    const start = { workflowId: '103161', runId: 'r', status: 'queued' }
    expect(workflowSchemas.execute_workflow.output.safeParse(start).success).toBe(true)
    expect(workflowSchemas.get_workflow_run.output.safeParse({ run: { ...start, nodes: [] } }).success).toBe(false)
  })
  it('W08 alpha represents versions and unknown writes; workflow patch has no such fields', () => {
    expect(() => parseCanvasOperation({ action: 'node.add', expectedVersion: '8', node: { type: 'text' } })).not.toThrow()
    expect(workflowSchemas.apply_workflow_patch.input.safeParse({ workflowId: '103161', expectedVersion: '8', operations: [] }).success).toBe(false)
    const unknown = { status: 'submission_unknown', canvasId: '103161', phase: 'save', version: null, affectedNodeIds: ['n'], error: { code: 'SUBMISSION_UNKNOWN', message: 'read first', retry: 'never_auto' } }
    expect(() => parseCanvasOperationResult(unknown)).not.toThrow()
    expect(workflowSchemas.apply_workflow_patch.output.safeParse(unknown).success).toBe(false)
  })
  it('W09 workflow offers parameter error location which alpha Failure lacks', () => {
    expect(workflowTypes.toolError.safeParse({ code: 'INVALID_PARAMETER', message: '参数不合法', operationIndex: 2, path: '/operations/2/node/duration' }).success).toBe(true)
  })
  it('W10 workflow operation limit and unknown-effect semantics need explicit bounds', () => {
    expect(workflowSchemas.apply_workflow_patch.input.safeParse({ workflowId: '103161', operations: Array.from({ length: 1001 }, () => ({ op: 'rename_workflow', name: 'review' })) }).success).toBe(true)
    expect(workflowSchemas.apply_workflow_patch.output.safeParse({ applied: true, valid: false, createdNodes: [], createdEdges: [], errors: [{ code: 'x', message: 'x' }], warnings: [] }).success).toBe(true)
  })
})
