import { describe, expect, it } from 'vitest'
import { MockWorkflowBackend } from '../packages/insight-workflow-op/src/mock-workflows'
import { workflowSchemas } from '../packages/insight-workflow-protocol/src/workflow-schemas'

// Review-only probes: passing means the documented migration gap was reproduced.
function setup() {
  const backend = new MockWorkflowBackend()
  return { backend, workflowId: backend.create('review').id }
}
const rename = (name: string) => ({ op: 'rename_workflow' as const, name })
const textNode = (clientId: string) => ({ op: 'add_node' as const, clientId, node: { type: 'text' as const, model: 'mock-text' } })
const ref = (clientId: string) => ({ kind: 'created' as const, clientId })
describe('review-only migration gap probes', () => {
  it('same idempotency key with a different payload returns old success', () => {
    const { backend, workflowId } = setup()
    backend.patch({ workflowId, idempotencyKey: 'k', operations: [rename('first')] })
    expect(backend.patch({ workflowId, idempotencyKey: 'k', operations: [rename('second')] }).applied).toBe(true)
    expect(backend.get(workflowId).name).toBe('first')
  })
  it('dryRun reusing a committed key returns applied=true', () => {
    const { backend, workflowId } = setup()
    const input = { workflowId, idempotencyKey: 'k', operations: [rename('first')] }
    backend.patch(input)
    expect(backend.patch({ ...input, dryRun: true }).applied).toBe(true)
  })
  it('accepts invented port ids, duplicate edges and a new cycle', () => {
    const { backend, workflowId } = setup()
    const edge = { op: 'add_edge' as const, clientId: 'ab', source: ref('a'), target: ref('b'), sourceHandle: 'invented', targetHandle: 'invented' }
    const result = backend.patch({ workflowId, operations: [textNode('a'), textNode('b'), edge, { ...edge, clientId: 'ab2' }, { ...edge, clientId: 'ba', source: ref('b'), target: ref('a') }] })
    expect(result.applied).toBe(true)
    expect(backend.get(workflowId).edges).toHaveLength(3)
  })
  it('creates nodes at the same coordinates and running without execution', () => {
    const { backend, workflowId } = setup()
    backend.patch({ workflowId, operations: [textNode('a'), textNode('b')] })
    const nodes = backend.get(workflowId).nodes
    expect(nodes.map(node => node.position)).toEqual([{ x: 0, y: 0 }, { x: 0, y: 0 }])
    expect(nodes.map(node => node.data.taskInfo.status)).toEqual(['running', 'running'])
  })
  it('patch contract rejects literal text content and expectedVersion', () => {
    expect(workflowSchemas.apply_workflow_patch.input.safeParse({ workflowId: 'w', operations: [{ ...textNode('a'), node: { ...textNode('a').node, text: 'saved copy' } }] }).success).toBe(false)
    expect(workflowSchemas.apply_workflow_patch.input.safeParse({ workflowId: 'w', expectedVersion: '1', operations: [rename('next')] }).success).toBe(false)
  })
  it('image mode update accepts a mode not supported by the chosen mock model', () => {
    const { backend, workflowId } = setup()
    const result = backend.patch({ workflowId, operations: [{ op: 'add_node', clientId: 'i', node: { type: 'image', model: 'mock-image', modelType: 'text_to_image' } }] })
    const input = { workflowId, operations: [{ op: 'update_node', nodeId: result.createdNodes[0].nodeId, changes: { type: 'image', modelType: 'reference_image_to_image' } }] }
    const parsed = workflowSchemas.apply_workflow_patch.input.parse(input)
    expect(backend.patch(parsed).applied).toBe(true)
  })
})
